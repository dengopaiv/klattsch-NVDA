# klattsch native engine.
#
# The product build: the library, the tools, and the stage verifiers as tests.
# Stage directories keep their own scratch scripts; this is what assembles the
# thing, from the same sources, on every platform it is built on.
#
# The exit test of stage 6 requires MSVC, clang-cl and gcc to produce identical
# samples, and that only works if it has been possible to run all three all
# along -- which is why this file exists at stage 1 rather than at the end.
#
#   cmake -B build -DCMAKE_BUILD_TYPE=Release
#   cmake --build build --config Release
#   ctest --test-dir build -C Release --output-on-failure

cmake_minimum_required(VERSION 3.16)

project(klattsch VERSION 0.8.0 LANGUAGES C)

# C17: C11 with its defect reports fixed, and the newest dialect every target
# compiler actually has. MSVC 19.51 has no /std:c23 at all, and its `clatest`
# is missing constexpr, nullptr, the bool keyword and #embed -- see
# docs/REWRITE.md, "Which C", and tools/probe-c23/ for the measurement.
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

# 64-bit only, per house rule 4 of the tree's CLAUDE.md. Fail loudly rather
# than quietly producing something nobody wants.
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR
    "32-bit builds are not produced here. Configure for x64 (or ARM64).")
endif()

# Warnings on, because a port is exactly where an implicit conversion or an
# uninitialised read hides.
if(MSVC)
  add_compile_options(/W4)
  # The dump tool writes binary to stdout and uses _setmode; MSVC's "secure
  # CRT" warnings about fwrite/fopen are noise for a build with no untrusted
  # input at all.
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Wshadow -Wconversion)
endif()

# --- the library ------------------------------------------------------------

add_library(klattsch STATIC
  csrc/kl_dsp.c
  csrc/kl_banks.c
  csrc/kl_banks_data.c      # generated by tools/build-banks-c.mjs
  csrc/kl_synth.c
  csrc/kl_norm.c
  csrc/kl_norm_data.c       # generated by tools/build-norm-c.mjs
  csrc/kl_token.c
  csrc/kl_compile.c
  csrc/kl_wav.c
  csrc/kl_render.c
  csrc/kl_fmt.c
  csrc/kl_text.c            # the text front end; see docs/19-frontend-text.md
  csrc/kl_text_rules.c
)
target_include_directories(klattsch PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/csrc)

# libm, where it is a separate library.
include(CheckLibraryExists)
check_library_exists(m sin "" HAVE_LIB_M)
if(HAVE_LIB_M)
  target_link_libraries(klattsch PUBLIC m)
endif()

# --- stage verification tools ----------------------------------------------

add_executable(kl_dsp_dump csrc/tools/kl_dsp_dump.c)
target_link_libraries(kl_dsp_dump PRIVATE klattsch)

add_executable(kl_banks_dump csrc/tools/kl_banks_dump.c)
target_link_libraries(kl_banks_dump PRIVATE klattsch)

add_executable(kl_synth_dump csrc/tools/kl_synth_dump.c)
target_link_libraries(kl_synth_dump PRIVATE klattsch)

add_executable(kl_norm_dump csrc/tools/kl_norm_dump.c)
target_link_libraries(kl_norm_dump PRIVATE klattsch)

add_executable(kl_token_dump csrc/tools/kl_token_dump.c)
target_link_libraries(kl_token_dump PRIVATE klattsch)

add_executable(kl_compile_dump csrc/tools/kl_compile_dump.c)
target_link_libraries(kl_compile_dump PRIVATE klattsch)

add_executable(kl_wav_dump csrc/tools/kl_wav_dump.c)
target_link_libraries(kl_wav_dump PRIVATE klattsch)

add_executable(kl_text_dump csrc/tools/kl_text_dump.c)
target_link_libraries(kl_text_dump PRIVATE klattsch)

# --- the program -----------------------------------------------------------

# The first artifact here that is a program rather than a test. It writes
# upstream's ISFT field into every file it renders; see csrc/kl_wav.h.
add_executable(klattsch_cli bin/klattsch_cli.c)
target_link_libraries(klattsch_cli PRIVATE klattsch)
set_target_properties(klattsch_cli PROPERTIES OUTPUT_NAME klattsch)
if(MINGW)
  # wmain needs the Unicode entry point; without this MinGW links the ANSI
  # one and the argument strings arrive in the system codepage.
  target_link_options(klattsch_cli PRIVATE -municode)
endif()

# --- the sample generator (Windows only) ------------------------------------

# docs/GENERATOR.md: a native Win32 window, one x64 executable with nothing to
# install beside it. C++ is used here and nowhere in the engine; the engine
# stays C17. See docs/20-generator.md.
#
# MSVC and clang-cl only. MinGW's resource compiler, windres, runs the
# preprocessor with the include directories unquoted, and this repository
# lives under a path with a space in it ("speech synthesis"): it fails before
# it reads the .rc. The GUI ships from MSVC, so the MinGW leg builds the
# engine and the tools and leaves the window out.
if(WIN32 AND MSVC)
  enable_language(CXX)
  enable_language(RC)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)

  add_executable(klattsch_gui WIN32
    gui-native/klattsch_gui.cpp
    gui-native/klattsch_gui.rc)
  target_compile_definitions(klattsch_gui PRIVATE
    UNICODE _UNICODE
    KL_VERSION_MAJOR=${PROJECT_VERSION_MAJOR}
    KL_VERSION_MINOR=${PROJECT_VERSION_MINOR}
    KL_VERSION_PATCH=${PROJECT_VERSION_PATCH}
    "KL_VERSION_STRING=\"${PROJECT_VERSION}\"")
  target_link_libraries(klattsch_gui PRIVATE comctl32 comdlg32 winmm shell32)

  # The static C runtime, so the exe runs on a Windows with no Visual C++
  # redistributable installed. The engine library the tools link keeps the
  # default runtime -- it is what four toolchains verified -- so the GUI
  # gets its own static-runtime build of the same sources.
  get_target_property(KL_ENGINE_SOURCES klattsch SOURCES)
  add_library(klattsch_mt STATIC ${KL_ENGINE_SOURCES})
  target_include_directories(klattsch_mt PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/csrc)
  set_target_properties(klattsch_mt klattsch_gui PROPERTIES
    MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_link_libraries(klattsch_gui PRIVATE klattsch_mt)
  # The manifest is in the .rc; stop the linker writing a second one.
  target_link_options(klattsch_gui PRIVATE /MANIFEST:NO)
endif()

# --- tests ------------------------------------------------------------------

include(CTest)

find_program(NODE_EXECUTABLE NAMES node nodejs)
if(NODE_EXECUTABLE)
  # Stage 1: the DSP primitives against the JavaScript they came from.
  add_test(NAME stage1-dsp
    COMMAND ${NODE_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tools/verify-stage1.mjs
            $<TARGET_FILE:kl_dsp_dump>)
  set_tests_properties(stage1-dsp PROPERTIES LABELS "validation;stage1")

  # The goldens must match the engine for any stage comparison to mean
  # anything, so this is a precondition rather than a separate concern.
  add_test(NAME goldens-current
    COMMAND ${NODE_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/goldens.mjs --check)
  set_tests_properties(goldens-current PROPERTIES LABELS "quick")

  add_test(NAME banks-current
    COMMAND ${NODE_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/build-banks.js --check)
  set_tests_properties(banks-current PROPERTIES LABELS "quick")

  # Stage 2: the compiled-in banks against the JS's resolved banks.
  add_test(NAME stage2-banks
    COMMAND ${NODE_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tools/verify-stage2.mjs
            $<TARGET_FILE:kl_banks_dump>)
  set_tests_properties(stage2-banks PROPERTIES LABELS "validation;stage2")

  # Stage 3: the sample loop, driven by the golden schedules.
  add_test(NAME stage3-synth
    COMMAND ${NODE_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tools/verify-stage3.mjs
            $<TARGET_FILE:kl_synth_dump>)
  set_tests_properties(stage3-synth PROPERTIES LABELS "validation;stage3")

  # Rendering in chunks must give the same samples as rendering whole. The
  # chunked-rendering work in phase 3 depends on it, so it is proved rather
  # than assumed.
  add_test(NAME stage3-chunked
    COMMAND ${NODE_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tools/verify-stage3.mjs
            $<TARGET_FILE:kl_synth_dump> --rate 48000 --chunked 137)
  set_tests_properties(stage3-chunked PROPERTIES LABELS "validation;stage3")

  # Stage 4: the tokenizer, plus the normalizer it sits on. The normalization
  # half is exhaustive -- every code point there is -- which is the shape of
  # test a singleton table deserves.
  add_test(NAME stage4-token
    COMMAND ${NODE_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tools/verify-stage4.mjs
            $<TARGET_FILE:kl_token_dump> $<TARGET_FILE:kl_norm_dump>)
  set_tests_properties(stage4-token PROPERTIES LABELS "validation;stage4")

  # Stage 5: the schedule compiler. Tier 1 -- every event time, transition and
  # target field compared as exact IEEE-754 doubles, plus the warning strings,
  # which are as much the contract as the numbers are.
  add_test(NAME stage5-compile
    COMMAND ${NODE_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tools/verify-stage5.mjs
            $<TARGET_FILE:kl_compile_dump>)
  set_tests_properties(stage5-compile PROPERTIES LABELS "validation;stage5")

  # Stage 6: the WAV encoder and the render-and-mix the CLI is built from,
  # over the whole corpus at all three sample rates. Whole files compared,
  # byte for byte -- the gain is a function of one sample of the mix, so a
  # single differing sample would move every byte in the file.
  add_test(NAME stage6-wav
    COMMAND ${NODE_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tools/verify-stage6.mjs
            $<TARGET_FILE:kl_wav_dump> --cli $<TARGET_FILE:klattsch_cli>)
  set_tests_properties(stage6-wav PROPERTIES LABELS "validation;stage6")

  # The text front end: the lift against Votraxxion's own output, the whole
  # front end against its captured output, and every line of that output
  # through the JavaScript compiler. The dictionary half of the lift is
  # checked only when cmu-pronouncing-dictionary is installed; see
  # docs/19-frontend-text.md.
  add_test(NAME text-frontend
    COMMAND ${NODE_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tools/verify-text.mjs
            $<TARGET_FILE:kl_text_dump>)
  set_tests_properties(text-frontend PROPERTIES LABELS "validation;text")

  # Pronunciation and stress measured against the CMU dictionary. Every count
  # must equal goldens/text/accuracy.json: a moved number is a changed rule,
  # re-measured on purpose with --update. Skipped (77) without the dictionary.
  add_test(NAME text-accuracy
    COMMAND ${NODE_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tools/measure-text.mjs
            $<TARGET_FILE:kl_text_dump> --check)
  set_tests_properties(text-accuracy PROPERTIES LABELS "validation;text"
                                                SKIP_RETURN_CODE 77)

  # The NFKC table is generated from Node's ICU, so it can go stale when Node
  # moves under it. Same guard as the banks, for the same reason.
  add_test(NAME norm-c-current
    COMMAND ${NODE_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/build-norm-c.mjs --check)
  set_tests_properties(norm-c-current PROPERTIES LABELS "quick")

  # The generated C bank table must not drift from the JSON it comes from --
  # the same guard bundled.js has, for the other consumer of the same source.
  add_test(NAME banks-c-current
    COMMAND ${NODE_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/build-banks-c.mjs --check)
  set_tests_properties(banks-c-current PROPERTIES LABELS "quick")
else()
  message(WARNING "node not found: the stage verifiers compare against the "
                  "JavaScript reference and will not be registered")
endif()
